          var params = new URLSearchParams(window.location.search)
          var forced = params.get('theme')
          var theme = forced === 'light' || forced === 'dark' ? forced : null
          if (!theme) {
            var stored = null
            try {
              stored = window.localStorage.getItem('theme')
            } catch (storageError) {
              stored = null
            }
            theme = stored === 'light' || stored === 'dark' ? stored : isStudioIframe ? 'light' : 'dark'
          }
          root.classList.remove('light', 'theme_dark')
          root.classList.add(theme === 'dark' ? 'theme_dark' : 'light')
          root.setAttribute('data-theme', theme)
          root.style.colorScheme = theme
        } catch (error) {
          console.error(error)
        }
      })()
